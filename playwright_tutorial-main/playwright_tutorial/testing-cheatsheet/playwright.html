import { test, expect } from '@playwright/test';

test.describe('Login', () => {
  test.beforeEach(async ({ page }) => {
    await page.goto('/login');
  });

  test('success', { tag: '@smoke' }, async ({ page }) => {

  });

  test.skip('legacy flow', async ({ page }) => { /* ... */ });
  test.fixme('flaky flow', async ({ page }) => { /* ... */ });
});

5.5.7 Test Structure

โครง test: describe, hooks, step, skip/fixme/only, fail, tag, data-driven, test.use

tests/login.spec.ts
import { test, expect } from '@playwright/test';

test.describe('Login', () => {
  test.beforeEach(async ({ page }) => {
    await page.goto('https://www.saucedemo.com');
  });

  test('success login', async ({ page }) => {
    await page.getByPlaceholder('Username').fill('standard_user');
    await page.getByPlaceholder('Password').fill('secret_sauce');
    await page.getByRole('button', { name: /login/i }).click();
    await expect(page).toHaveURL(/inventory/);
  });

  test('failed login shows error', async ({ page }) => {
    await page.getByPlaceholder('Username').fill('standard_user');
    await page.getByPlaceholder('Password').fill('wrong');
    await page.getByRole('button', { name: /login/i }).click();
    await expect(page.getByRole('alert')).toBeVisible();
  });

  test.skip('legacy flow', async ({ page }) => { /* ... */ });
  test.fixme('flaky flow', async ({ page }) => { /* ... */ });
});

test('data-driven loop', async ({ page }) => {
  const users = [
    { username: 'standard_user', password: 'secret_sauce', expected: 'inventory' },
    { username: 'locked_out_user', password: 'wrong', expected: 'error' },
  ];
  for (const u of users) {
    await page.getByPlaceholder('Username').fill(u.username);
    await page.getByPlaceholder('Password').fill(u.password);
    await page.getByRole('button', { name: /login/i }).click();
    if (u.expected === 'inventory') {
      await expect(page).toHaveURL(new RegExp(u.expected));
    } else {
      await expect(page.getByRole('alert')).toBeVisible();
    }
  }
});

test.use({ viewport: { width: 375, height: 667 } });

5.5.9 Page Object Model (POM)

โครง pages/LoginPage.ts, pages/DashboardPage.ts และเทสต์ที่ใช้ POM

tests/pages/LoginPage.ts
import type { Page } from '@playwright/test';

export class LoginPage {
  readonly page: Page;
  readonly username = this.page.getByPlaceholder('Username');
  readonly password = this.page.getByPlaceholder('Password');
  readonly submit = this.page.getByRole('button', { name: /login/i });

  constructor(page: Page) {
    this.page = page;
  }

  async login(user: string, pass: string) {
    await this.page.goto('https://www.saucedemo.com');
    await this.username.fill(user);
    await this.password.fill(pass);
    await this.submit.click();
  }
}
tests/login-pom.spec.ts
import { test, expect } from '@playwright/test';
import { LoginPage } from './pages/LoginPage';

test('POM login flow', async ({ page }) => {
  const login = new LoginPage(page);
  await login.login('standard_user', 'secret_sauce');
  await expect(page).toHaveURL(/inventory/);
});

สรุปท่องจำ (Key Points)

  • ข้อดีของ POM: ลดโค้ดซ้ำ, แก้ที่เดียว, อ่านง่าย
  • เทียบกับแบบไม่ใช้ POM: โค้ด test สั้นขึ้น แต่มี class boilerplate

5.5.11 API Testing

ใช้ request fixture: GET/POST/PUT/DELETE, ตรวจ status, ตรวจ body ด้วย expect

tests/api.spec.ts
import { test, expect } from '@playwright/test';

test.use({ baseURL: 'https://jsonplaceholder.typicode.com' });

test('create and read post via API', async ({ request }) => {
  const create = await request.post('/posts', {
    data: { title: 'Hello', body: 'World', userId: 1 },
  });
  expect(create.ok()).toBeTruthy();
  const created = await create.json();
  expect(created.id).toBeDefined();

  const get = await request.get(`/posts/${created.id}`);
  expect(get.ok()).toBeTruthy();
  const body = await get.json();
  expect(body.title).toBe('Hello');
});

test('update post', async ({ request }) => {
  const res = await request.put('/posts/1', { data: { title: 'Updated', body: 'B', userId: 1 } });
  expect(res.status()).toBe(200);
});

test('delete post', async ({ request }) => {
  const res = await request.delete('/posts/1');
  expect(res.status()).toBe(200);
});

ตัวอย่างเต็วทดสอบ CRUD กับ public API — ตรวจความพร้อมใช้งานก่อนเลือก (เช่น https://jsonplaceholder.typicode.com, https://reqres.in)

Send headers/token, data / form / multipart, ผสม API + UI:

  await page.goto('https://demo.playwright.dev/todomvc');
  await page.getByPlaceholder('What needs to be done?').fill(todo.title);
  await page.keyboard.press('Enter');
  await expect(page.getByRole('listitem').filter({ hasText: todo.title })).toHaveCount(1);
});

5.5.15 Authentication & Storage State

login ครั้งเดียว → บันทึก storageState → ใช้ซ้ำ

tests/global-setup.ts
import { test as base } from '@playwright/test';

export const test = base.extend({
  authenticatedPage: async ({ page }, use, testInfo) => {
    // login once and save storage state
    await page.goto('https://www.saucedemo.com');
    await page.getByPlaceholder('Username').fill('standard_user');
    await page.getByPlaceholder('Password').fill('secret_sauce');
    await page.getByRole('button', { name: /login/i }).click();
    await page.context().storageState({ path: testInfo.outputPath('auth.json') });
    await use(page);
  },
});

ใน config นำ state มาใช้:

playwright.config.ts
use: { storageState: 'playwright/.auth/user.json' }

5.5.18 Best Practices

สรุปท่องจำ (Key Points)

  • ทดสอบแบบที่ผู้ใช้เห็น — test user flow จริง
  • test ต้อง isolated, ใช้ locator ที่ทนทาน
  • ห้ามใช้ waitForTimeout / sleep — ใช้ web-first assertions
  • mock third-party API ใน test ที่ไม่ต้องการ external
  • CI พัง → ใช้ trace
